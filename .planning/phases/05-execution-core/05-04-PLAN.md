    **Step 2 — Refactor the spawn site in `run_one`.** Plan 03's `run_one` currently calls `spawn_command(&task.argv, &task.working_dir).await` directly. Plan 04 splits this into a **manual** spawn so the `Child` handle is available for kill, then either awaits `child.wait_with_output()` directly OR wraps it in `tokio::time::timeout(d, ...)`:

    ```rust
    use tokio::process::Command;
    use tokio::time::{timeout, Duration as TokioDuration};
    use std::process::Stdio;

    // Inside run_one(task: Task, cancel_flag: Arc<AtomicBool>, effective_timeout: Option<Duration>) -> CommandOutcome
    // (signature gains a third parameter — see Step 3)

    if cancel_flag.load(Ordering::Relaxed) {
        return CommandOutcome::Cancelled { /* ... */ };
    }
    let started = std::time::Instant::now();

    // Spawn manually so we keep the Child handle for kill.
    let mut child = match Command::new(&task.argv[0])
        .args(&task.argv[1..])
        .current_dir(&task.working_dir)
        .stdout(Stdio::piped())
        .stderr(Stdio::piped())
        .spawn()
    {
        Ok(c) => c,
        Err(e) if e.kind() == std::io::ErrorKind::NotFound => {
            return failure_outcome_command_not_found(&task, started.elapsed());
        }
        Err(e) => {
            return failure_outcome_io(&task, started.elapsed(), e);
        }
    };

    // Capture pid BEFORE awaiting (Pitfall 2 — pid may be reaped post-wait on some platforms).
    let pid_for_kill: Option<u32> = child.id();

    let wait_future = child.wait_with_output();

    let output_result: Result<std::process::Output, std::io::Error> = match effective_timeout {
        Some(d) => match timeout(d, wait_future).await {
            Ok(inner) => inner,
            Err(_elapsed) => {
                // Timeout: terminate child via cross-platform helper.
                // wait_with_output consumed `child`, so terminate must take pid only.
                terminate_by_pid(pid_for_kill).await;
                return failure_outcome_timeout(&task, started.elapsed(), d);
            }
        },
        None => wait_future.await,
    };
    ```

    **CRITICAL CORRECTION** — `child.wait_with_output()` consumes `child` by value. To call `child.kill()` on timeout, the spawn must NOT use `wait_with_output()`. Use this shape instead:

    ```rust
    let mut child = /* spawn as above */;

    // We'll await wait()+drain stdout/stderr ourselves so we keep the Child handle for kill.
    let stdout_handle = child.stdout.take();
    let stderr_handle = child.stderr.take();

    let wait_fut = child.wait();

    let exit_status = match effective_timeout {
        Some(d) => match timeout(d, wait_fut).await {
            Ok(inner) => inner,
            Err(_elapsed) => {
                // Unix: SIGTERM, wait up to 2s, then SIGKILL via child.kill().await.
                #[cfg(unix)]
                {
                    if let Some(pid) = child.id() {
                        let _ = nix::sys::signal::kill(
                            nix::unistd::Pid::from_raw(pid as i32),
                            nix::sys::signal::Signal::SIGTERM,
                        );
                        let _ = timeout(TokioDuration::from_secs(2), child.wait()).await;
                    }
                }
                // Force kill (SIGKILL on Unix, TerminateProcess on Windows). Reaps child too.
                let _ = child.kill().await;
                return failure_outcome_timeout(&task, started.elapsed(), d);
            }
        },
        None => wait_fut.await,
    };

    let exit_status = match exit_status {
        Ok(s) => s,
        Err(e) => return failure_outcome_io(&task, started.elapsed(), e),
    };

    // Drain captured stdout/stderr buffers. Use AsyncReadExt::read_to_end.
    use tokio::io::AsyncReadExt;
    let mut stdout_buf = Vec::new();
    if let Some(mut h) = stdout_handle { let _ = h.read_to_end(&mut stdout_buf).await; }
    let mut stderr_buf = Vec::new();
    if let Some(mut h) = stderr_handle { let _ = h.read_to_end(&mut stderr_buf).await; }

    let elapsed = started.elapsed();
    if exit_status.success() {
        CommandOutcome::Success {
            group: task.group_name,
            pattern: task.pattern,
            command: task.command,
            argv: task.argv,
            file_count: task.file_count,
            duration: elapsed,
            stdout: stdout_buf,
        }
    } else {
        CommandOutcome::Failure {
            group: task.group_name,
            pattern: task.pattern,
            command: task.command,
            argv: task.argv,
            file_count: task.file_count,
            duration: elapsed,
            exit_code: exit_status.code().unwrap_or(-1),
            stdout: stdout_buf,
            stderr: stderr_buf,
        }
    }
    ```

    fn failure_outcome_timeout(task: &Task, duration: std::time::Duration, configured: std::time::Duration) -> CommandOutcome {
