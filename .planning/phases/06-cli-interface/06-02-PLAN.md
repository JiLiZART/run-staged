```rust
#[test]
fn config_explicit_loads() {
    let dir = make_real_git_repo("e2e-cli-explicit-config");
    // Stage a file matching the explicit config's pattern so the run reaches command exec.
    stage_file(&dir, "a.txt", "hello\n");
    // Write the config to a NON-default name + nested path so auto-discovery would NOT find it.
    std::fs::create_dir_all(dir.join("nested")).unwrap();
    write_config(
        &dir,
        "nested/custom.toml",
        r#"
            [group1.patterns]
            "*.txt" = ["echo"]
        "#,
    );
    let explicit = dir.join("nested").join("custom.toml");
    let out = run_binary(&dir, &["--config", explicit.to_str().unwrap(), "--allow-empty"]);
    assert_eq!(
        out.status.code(),
        Some(0),
        "explicit --config should load and exit 0; stdout={} stderr={}",
        String::from_utf8_lossy(&out.stdout),
        String::from_utf8_lossy(&out.stderr)
    );
}

#[test]
fn config_missing_exits_nonzero() {
    let dir = make_real_git_repo("e2e-cli-missing-config");
    // Pass an absolute path that definitely does not exist.
    let bogus = dir.join("does-not-exist").join("config.toml");
    let out = run_binary(&dir, &["--config", bogus.to_str().unwrap()]);
    assert_eq!(
        out.status.code(),
        Some(1),
        "missing --config path should exit 1 (AppError::ConfigNotFound); stdout={} stderr={}",
        String::from_utf8_lossy(&out.stdout),
        String::from_utf8_lossy(&out.stderr)
    );
    let stderr = String::from_utf8_lossy(&out.stderr);
    assert!(
        stderr.contains("configuration file not found"),
        "stderr should contain the AppError::ConfigNotFound display string; got:\n{stderr}"
    );
}

#[test]
fn quiet_flag_suppresses_output() {
    let dir = make_real_git_repo("e2e-cli-quiet");
    stage_file(&dir, "a.txt", "hello\n");
    write_config(
        &dir,
        ".fast-staged.toml",
        r#"
            [group1.patterns]
            "*.txt" = ["echo"]
        "#,
    );
    let out = run_binary(&dir, &["--quiet"]);
    assert_eq!(
        out.status.code(),
        Some(0),
        "successful run with --quiet should exit 0; stdout={} stderr={}",
        String::from_utf8_lossy(&out.stdout),
        String::from_utf8_lossy(&out.stderr)
    );
    let stdout = String::from_utf8_lossy(&out.stdout);
    assert!(
        stdout.trim().is_empty(),
        "--quiet should suppress success summary; stdout was:\n{stdout}"
    );
}
```

Notes:
- `config_explicit_loads`: places the config in a nested dir with a non-standard filename. Without `--config`, auto-discovery (which only checks the 5 standard names — `.fast-staged.toml`, `.fast-staged.json`, `fast-staged.toml`, `fast-staged.json`, `package.json`) would fail with `ConfigNotFound`. The test passing therefore proves `--config` overrides auto-discovery.
- `config_missing_exits_nonzero`: D-02/D-03 — exit 1 (our `AppError::ConfigNotFound`), NOT exit 2 (clap parse error). The path is syntactically valid; only the file is absent. The error message check uses the existing `ConfigNotFound` display string from `error.rs:5`.
- `quiet_flag_suppresses_output`: Verifies CLI-01 wiring of `--quiet` end-to-end (Phase 5 already implemented `quiet` behavior in `output::print_results`; this test confirms the CLI flag still reaches it after the clap refactor).
- All three tests use unique `make_real_git_repo` suffixes to enable parallel test execution per the file's lead comment.
